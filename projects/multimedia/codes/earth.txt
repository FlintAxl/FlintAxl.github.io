import bpy
import math

# --- Clear All Objects ---
bpy.ops.object.select_all(action='SELECT')
bpy.ops.object.delete(use_global=False)

# --- Create Earth ---
bpy.ops.mesh.primitive_uv_sphere_add(radius=2, location=(0, 0, 0))
earth = bpy.context.object
earth.name = "Earth"

bpy.ops.object.shade_smooth()
subsurf = earth.modifiers.new(name="Subdivision", type='SUBSURF')
subsurf.levels = 2
subsurf.render_levels = 3

# --- Earth Material with Texture ---
earth_mat = bpy.data.materials.new(name="EarthMaterial")
earth_mat.use_nodes = True
nodes = earth_mat.node_tree.nodes
links = earth_mat.node_tree.links
nodes.clear()

output_node = nodes.new(type='ShaderNodeOutputMaterial')
bsdf_node = nodes.new(type='ShaderNodeBsdfPrincipled')
tex_image = nodes.new(type='ShaderNodeTexImage')
tex_coord = nodes.new(type='ShaderNodeTexCoord')
mapping = nodes.new(type='ShaderNodeMapping')

# Load Earth texture
img_path = r"C:\Users\THINKPAD PC\Downloads\2k_sun.jpg"
tex_image.image = bpy.data.images.load(img_path)

# Connect Earth nodes
links.new(tex_coord.outputs['Generated'], mapping.inputs['Vector'])
links.new(mapping.outputs['Vector'], tex_image.inputs['Vector'])
links.new(tex_image.outputs['Color'], bsdf_node.inputs['Base Color'])
links.new(bsdf_node.outputs['BSDF'], output_node.inputs['Surface'])

earth.data.materials.append(earth_mat)

# --- Add Lighting ---
bpy.ops.object.light_add(type='SUN', location=(10, 10, 10))
light = bpy.context.object
light.data.energy = 3
light.data.use_shadow = True

# --- Add Camera ---
bpy.ops.object.camera_add(location=(6, -6, 4.4))
camera = bpy.context.object
camera.rotation_euler = (math.radians(65), 0, math.radians(45))
bpy.context.scene.camera = camera

# --- Animate Earth's Rotation ---
earth.rotation_euler = (0, 0, 0)
earth.keyframe_insert(data_path="rotation_euler", frame=1)
earth.rotation_euler = (0, 0, math.radians(360))
earth.keyframe_insert(data_path="rotation_euler", frame=250)
bpy.context.scene.frame_start = 1
bpy.context.scene.frame_end = 250
bpy.context.scene.render.fps = 24

# --- Add Background Environment Image ---
bpy.context.scene.render.engine = 'BLENDER_EEVEE_NEXT'  # or 'CYCLES'

world = bpy.context.scene.world
world.use_nodes = True
wnodes = world.node_tree.nodes
wlinks = world.node_tree.links
wnodes.clear()

env_node = wnodes.new(type='ShaderNodeTexEnvironment')
bg_node = wnodes.new(type='ShaderNodeBackground')
out_node = wnodes.new(type='ShaderNodeOutputWorld')

# Load background image
bg_path = r"C:\Users\THINKPAD PC\Downloads\beautiful-view-stars-night-sky.jpg"
try:
    env_node.image = bpy.data.images.load(bg_path)
except:
    print("❌ Failed to load background image")

# Connect nodes
wlinks.new(env_node.outputs['Color'], bg_node.inputs['Color'])
wlinks.new(bg_node.outputs['Background'], out_node.inputs['Surface'])

# Optional: Layout
env_node.location = (-300, 0)
bg_node.location = (0, 0)
out_node.location = (300, 0)

# Enable World Background in viewport shading
for area in bpy.context.screen.areas:
    if area.type == 'VIEW_3D':
        for space in area.spaces:
            if space.type == 'VIEW_3D':
                space.shading.use_scene_world = True
                space.shading.type = 'RENDERED'

print("✅ Earth created with background and animation.")
