import bpy
import math
import os
import mathutils
from mathutils import Matrix


# --------------------------------------------------
# Helper: Create or get a collection
# --------------------------------------------------
def get_collection(name):
    if name in bpy.data.collections:
        return bpy.data.collections[name]
    coll = bpy.data.collections.new(name)
    bpy.context.scene.collection.children.link(coll)
    return coll

# --------------------------------------------------
# Clear existing solar system
# --------------------------------------------------
def clear_scene():
    bpy.ops.object.select_all(action='SELECT')
    bpy.ops.object.delete(use_global=False)
    for coll in list(bpy.data.collections):
        if coll.name.startswith("SolarSystem"):
            bpy.data.collections.remove(coll)

# --------------------------------------------------
# Set space background image and enable in preview
# --------------------------------------------------
def set_space_background(image_path):
    world = bpy.context.scene.world
    if world is None:
        world = bpy.data.worlds.new("World")
        bpy.context.scene.world = world

    world.use_nodes = True
    nodes = world.node_tree.nodes
    links = world.node_tree.links
    nodes.clear()

    tex_node = nodes.new(type='ShaderNodeTexEnvironment')
    bg_node = nodes.new(type='ShaderNodeBackground')
    output_node = nodes.new(type='ShaderNodeOutputWorld')

    try:
        tex_node.image = bpy.data.images.load(image_path)
    except Exception as e:
        print(f"Failed to load background image: {image_path}: {e}")
        return

    links.new(tex_node.outputs['Color'], bg_node.inputs['Color'])
    links.new(bg_node.outputs['Background'], output_node.inputs['Surface'])

    for area in bpy.context.screen.areas:
        if area.type == 'VIEW_3D':
            for space in area.spaces:
                if space.type == 'VIEW_3D':
                    space.shading.use_scene_world = True

# --------------------------------------------------
# Create Sun with texture emission
# --------------------------------------------------
def create_sun(collection, texture_path):
    bpy.ops.mesh.primitive_uv_sphere_add(segments=64, ring_count=32, radius=1, location=(0,0,0))
    sun = bpy.context.active_object
    sun.name = "Sun"

    mat = bpy.data.materials.new(name="SunMaterial")
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    links = mat.node_tree.links
    nodes.clear()

    tex_node = nodes.new('ShaderNodeTexImage')
    try:
        tex_node.image = bpy.data.images.load(texture_path)
    except Exception as e:
        print(f"Failed to load sun texture: {texture_path}: {e}")

    emission = nodes.new('ShaderNodeEmission')
    links.new(tex_node.outputs['Color'], emission.inputs['Color'])
    emission.inputs['Strength'].default_value = 5.0

    output = nodes.new('ShaderNodeOutputMaterial')
    links.new(emission.outputs['Emission'], output.inputs['Surface'])

    sun.data.materials.append(mat)

    bpy.ops.object.light_add(type='POINT', radius=1, location=(0,0,0))
    light = bpy.context.active_object
    light.name = "SunLight"
    light.data.energy = 5000
    light.data.shadow_soft_size = 0.5

    collection.objects.link(sun)
    collection.objects.link(light)

# --------------------------------------------------
# Create orbit material (white emission)
# --------------------------------------------------
def create_orbit_material():
    mat = bpy.data.materials.new(name="OrbitWhiteMaterial")
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    links = mat.node_tree.links
    nodes.clear()

    emission = nodes.new('ShaderNodeEmission')
    emission.inputs['Color'].default_value = (1, 1, 1, 1)  # Pure white
    emission.inputs['Strength'].default_value = 3.0  # Increased strength for better visibility

    output = nodes.new('ShaderNodeOutputMaterial')
    links.new(emission.outputs['Emission'], output.inputs['Surface'])

    # Enable backface culling and make material visible in viewport
    mat.use_backface_culling = True
    mat.blend_method = 'BLEND'
    mat.shadow_method = 'NONE'
    
    return mat

# --------------------------------------------------
# Create a planet with animation and label
# --------------------------------------------------
def create_planet(collection, name, radius, texture_path, distance, axial_tilt, self_days, orbit_days):
    bpy.ops.mesh.primitive_uv_sphere_add(segments=32, ring_count=16, radius=radius, location=(distance,0,0))
    planet = bpy.context.active_object
    planet.name = name
    planet.rotation_euler = (math.radians(axial_tilt), 0, 0)

    mat = bpy.data.materials.new(name=f"{name}Mat")
    mat.use_nodes = True
    nodes = mat.node_tree.nodes
    links = mat.node_tree.links
    nodes.clear()

    tex = nodes.new('ShaderNodeTexImage')
    try:
        tex.image = bpy.data.images.load(texture_path)
    except:
        bsdf = nodes.new('ShaderNodeBsdfPrincipled')
        bsdf.inputs['Base Color'].default_value = (0.8,0.8,0.8,1)
        out = nodes.new('ShaderNodeOutputMaterial')
        links.new(bsdf.outputs['BSDF'], out.inputs['Surface'])
        planet.data.materials.append(mat)
        collection.objects.link(planet)
        return

    normal = nodes.new('ShaderNodeNormalMap')
    bsdf = nodes.new('ShaderNodeBsdfPrincipled')
    bsdf.inputs['Roughness'].default_value = 0.9
    out = nodes.new('ShaderNodeOutputMaterial')

    links.new(tex.outputs['Color'], bsdf.inputs['Base Color'])
    links.new(normal.outputs['Normal'], bsdf.inputs['Normal'])
    links.new(bsdf.outputs['BSDF'], out.inputs['Surface'])

    planet.data.materials.append(mat)
    
    

    bpy.ops.object.empty_add(type='PLAIN_AXES', location=(0,0,0))
    orbit_empty = bpy.context.active_object
    orbit_empty.name = f"{name}_Orbit"
    planet.parent = orbit_empty

    orbit_empty.rotation_mode = 'XYZ'
    orbit_empty.keyframe_insert(data_path='rotation_euler', frame=1)
    orbit_empty.rotation_euler.z = math.radians(360)
    orbit_empty.keyframe_insert(data_path='rotation_euler', frame=orbit_days)

    for fc in orbit_empty.animation_data.action.fcurves:
        fc.extrapolation = 'LINEAR'
        for kp in fc.keyframe_points:
            kp.interpolation = 'LINEAR'

    planet.keyframe_insert(data_path='rotation_euler', frame=1)
    planet.rotation_euler.z += math.radians(360)
    planet.keyframe_insert(data_path='rotation_euler', frame=max(1,int(self_days)))

    for fc in planet.animation_data.action.fcurves:
        fc.extrapolation = 'LINEAR'
        for kp in fc.keyframe_points:
            kp.interpolation = 'LINEAR'

    bpy.ops.curve.primitive_bezier_circle_add(radius=distance, location=(0,0,0))
    path = bpy.context.active_object
    path.name = f"{name}_OrbitPath"
    path.data.bevel_depth = 0.002  # Slightly thicker orbit line
    path.data.resolution_u = 128  # Higher resolution for smoother circle
    path.display_type = 'WIRE'

    
    # Apply enhanced white material to orbit path
    orbit_mat = bpy.data.materials.get("OrbitWhiteMaterial") or create_orbit_material()
    path.data.materials.append(orbit_mat)

    bpy.ops.object.text_add(location=(distance, 0, radius + 0.5))
    label = bpy.context.active_object
    label.name = f"{name}_Label"
    label.data.body = name
    label.data.size = 1.5
    label.data.align_x = 'CENTER'
    label.data.extrude = 0.05
    label.data.bevel_depth = 0.01
    label.rotation_euler = (math.radians(90), 0, 0)
    label.parent = orbit_empty

    constraint = label.constraints.new(type='TRACK_TO')
    constraint.target = bpy.context.scene.camera
    constraint.track_axis = 'TRACK_Z'
    constraint.up_axis = 'UP_Y'

    text_mat = bpy.data.materials.new(name=f"{name}_LabelMat")
    text_mat.use_nodes = True
    nodes = text_mat.node_tree.nodes
    links = text_mat.node_tree.links
    nodes.clear()

    emission = nodes.new('ShaderNodeEmission')
    emission.inputs['Color'].default_value = (1, 1, 1, 1)
    emission.inputs['Strength'].default_value = 10

    output = nodes.new('ShaderNodeOutputMaterial')
    links.new(emission.outputs['Emission'], output.inputs['Surface'])

    label.data.materials.append(text_mat)

    for obj in (orbit_empty, planet, path, label):
        collection.objects.link(obj)
        
        
            # Add Saturn's ring
    if name == "Saturn":
        bpy.ops.mesh.primitive_torus_add(location=(distance, 0, 0), major_radius=radius + 0.8, minor_radius=0.2, major_segments=48, minor_segments=12)
        ring = bpy.context.active_object
        ring.name = "SaturnRing"
        ring.rotation_euler = (math.radians(180), 0, 0)
        ring.parent = orbit_empty

        # Ring material
        ring_mat = bpy.data.materials.new(name="SaturnRingMat")
        ring_mat.use_nodes = True
        nodes = ring_mat.node_tree.nodes
        links = ring_mat.node_tree.links
        nodes.clear()

        emission = nodes.new('ShaderNodeEmission')
        emission.inputs['Color'].default_value = (0.8, 0.7, 0.5, 1)
        emission.inputs['Strength'].default_value = 1.0

        output = nodes.new('ShaderNodeOutputMaterial')
        links.new(emission.outputs['Emission'], output.inputs['Surface'])

        ring.data.materials.append(ring_mat)
        collection.objects.link(ring)
